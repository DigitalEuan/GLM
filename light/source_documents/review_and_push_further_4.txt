def run_obstacle_simulation():
    print("=" * 80)
    print(" UBP LIGHT-OBSTACLE INTERACTION SIMULATION")
    print("=" * 80)

    # 1. Base Photon (Golay Octad)
    octads = GOLAY_ENGINE.get_octads()
    photon_vec = octads[0]
    tax_vacuum = LEECH_ENGINE.calculate_symmetry_tax(photon_vec)

    tax_medium = LEECH_ENGINE.calculate_symmetry_tax(obstacle_vec)

    print(f"Vacuum Photon Tax : {float(tax_vacuum):.4f} (NRCI = {float(nrci_vacuum):.4f})")

    print("-" * 80)

    # 3. Simulate Propagation across 12 Ticks
    x_pos = 0.0
    spheres = []
    lines = []
    path_points = []

    # Render Medium Barrier in 3D Scene
    for bx in [4.0, 5.5, 7.0]:
        for by in [-2.0, 0.0, 2.0]:
            for bz in [-2.0, 0.0, 2.0]:
                spheres.append({
                    "x": bx, "y": by, "z": bz,
                    "r": 0.35,
                    "color": "#3366ff",  # Blue for medium/barrier
                    "label": "Medium_Barrier"
                })

    print(f"{'Tick':<6} | {'X Pos':<8} | {'Phase Shift (Δφ)':<18} | {'Speed (v/c)':<12} | {'Tax':<10} | {'Region'}")
    print("-" * 80)

    for tick in range(1, 13):
        # Determine if inside medium
        in_medium = (4.0 <= x_pos <= 7.0)

        if in_medium:
            # Interaction inside medium induces phase shift Δφ = 48° (48-degree anomaly)
            phase_deg = 48.0
            region_name = "MEDIUM (Refracted)"
            color = "#00ffcc"  # Cyan in medium
        else:
            phase_deg = 90.0
            region_name = "VACUUM (Free)"
            color = "#ffff00"  # Bright yellow in vacuum

        rad = math.radians(phase_deg)
        v_step = math.sin(rad)  # v = c * sin(Δφ)
        
        # Calculate local tax
        drift = 1.0 - math.sin(rad)
        current_tax = float(tax_vacuum) + (drift * float(LEECH_ENGINE.Y) * 12.0)

        # Update position
        x_pos += v_step
        py = math.sin(x_pos * 0.8) * 1.5 if not in_medium else math.sin(x_pos * 1.6) * 0.8
        pz = math.cos(x_pos * 0.8) * 1.5 if not in_medium else math.cos(x_pos * 1.6) * 0.8

        path_points.append([x_pos, py, pz])

        spheres.append({
            "x": x_pos, "y": py, "z": pz,
            "r": 0.25,
            "color": color,
            "label": f"Photon_t{tick}"
        })

        print(f"Tick {tick:02d} | {x_pos:>7.3f} | {phase_deg:>5.1f}°              | {v_step:>10.4f} | {current_tax:>8.4f} | {region_name}")

    # Build 3D Lines for Photon Trajectory
    for i in range(len(path_points) - 1):
        lines.append({
            "start": path_points[i],
            "end": path_points[i+1],
            "color": "#ffff00"
        })

    # Save 3D Scene
    scene = {"spheres": spheres, "lines": lines}
    with open("scene_3d.json", "w") as f:
        json.dump(scene, f, indent=2)

    # Refractive Index Calculation
    n_effective = 1.0 / math.sin(math.radians(48.0))
    print("-" * 80)
    print(f"Emergent Refractive Index of Substrate Medium (n = c / v) : {n_effective:.4f}")
    print("3D Scene saved to 'scene_3d.json'. Check the 'Visual' tab!")

if __name__ == "__main__":
    run_obstacle_simulation()
