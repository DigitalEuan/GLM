GF4_SYMBOLS = {0: "0", 1: "1", 2: "ω", 3: "ω̄"}
GF4_ADD = [

# 16-State Column Bijection: F_2^4 <-> GF(4) x Z_4
COLUMN_TO_SHADOW: Dict[Tuple[int, int, int, int], Tuple[int, int]] = {}
SHADOW_TO_COLUMN: Dict[Tuple[int, int], Tuple[int, int, int, int]] = {}

for _val in range(16):
    _b = ((_val >> 3) & 1, (_val >> 2) & 1, (_val >> 1) & 1, _val & 1)
    _s = 0
    for _r in range(4):
        if _b[_r]:
            _s = GF4_ADD[_s][ROW_W[_r]]
    _f = sum(1 for _c, (_sc, _) in COLUMN_TO_SHADOW.items() if _sc == _s)
    COLUMN_TO_SHADOW[_b] = (_s, _f)
    SHADOW_TO_COLUMN[(_s, _f)] = _b

def project_vector_to_mog(vec24: List[int]) -> Tuple[List[int], List[int]]:
    grid = [vec24[i * 6 : (i + 1) * 6] for i in range(4)]
    scores, fibers = [], []
    for c in range(6):
        col = tuple(grid[r][c] for r in range(4))
        s, f = COLUMN_TO_SHADOW[col]
        scores.append(s)
        fibers.append(f)
    return scores, fibers


def reconstruct_vector_from_mog(scores: List[int], fibers: List[int]) -> List[int]:
    grid = [[0] * 6 for _ in range(4)]
    for c in range(6):
        col = SHADOW_TO_COLUMN[(scores[c], fibers[c])]
        for r in range(4):
            grid[r][c] = col[r]
    return grid[0] + grid[1] + grid[2] + grid[3]

    fiber: List[int] = field(init=False)

    tax: float = field(init=False)

    def __post_init__(self):
        dims = self.dimensions
        # 4 Ontological Tiers (6 bits each = 24 bits)
        reality = [1 if dims[i] != 0 else 0 for i in range(6)]
        info = [dims[i] % 2 for i in range(5)] + [dims[6] % 2]
        activation = [1 if abs(dims[i]) > 1 else 0 for i in range(5)] + [1 if abs(dims[6]) > 1 else 0]
        potential = [1 if dims[i] < 0 else 0 for i in range(5)] + [1 if dims[6] < 0 else 0]

        snapped, _ = GOLAY_ENGINE.snap_to_codeword(raw_vec)
        self.vector_24 = snapped

        self.tax = float(LEECH_ENGINE.calculate_symmetry_tax(snapped))
        self.nrci = float(LEECH_ENGINE.calculate_nrci(snapped))

    def dims_str(self) -> str:
        parts = []
        for n, e in zip(DIM_NAMES, self.dimensions):
            if e == 1: parts.append(n)
            elif e != 0: parts.append(f"{n}^{e}")
        return "·".join(parts) if parts else "dimensionless"

class GeometricReasoner:
    def __init__(self):
        self.entities = {e.name.lower(): e for e in CANONICAL_ENTITIES}
        self.symbol_map = {e.symbol.lower(): e for e in CANONICAL_ENTITIES}

    def resolve_entity(self, key: str) -> Optional[GeometricEntity]:
        k = key.lower().strip()
        return self.entities.get(k) or self.symbol_map.get(k)

    def compose(self, a_name: str, b_name: str, op: str = "*") -> Optional[GeometricEntity]:
        """Composes two entities algebraically in (Z^7, +)."""
        ea = self.resolve_entity(a_name)
        eb = self.resolve_entity(b_name)
        if not ea or not eb:
            return None

        if op == "*":
            new_dims = [x + y for x, y in zip(ea.dimensions, eb.dimensions)]
            sym = f"({ea.symbol}·{eb.symbol})"
            name = f"({ea.name}*{eb.name})"
        elif op == "/":
            new_dims = [x - y for x, y in zip(ea.dimensions, eb.dimensions)]
            sym = f"({ea.symbol}/{eb.symbol})"
            name = f"({ea.name}/{eb.name})"
        else:
            raise ValueError(f"Unknown operation: {op}")

        return GeometricEntity(name=name, symbol=sym, dimensions=new_dims)

    def solve_relation(self, inputs: List[str], target: str) -> Dict[str, Any]:
        """
        Deduction Engine: Determines how a list of input quantities mathematically
        produces the target quantity using (Z^7, +) dimensional balancing.
        """
        target_entity = self.resolve_entity(target)
        if not target_entity:
            return {"status": "ERROR", "reason": f"Target entity '{target}' not in knowledge base."}

        resolved_inputs = [self.resolve_entity(i) for i in inputs if self.resolve_entity(i)]
        if len(resolved_inputs) != len(inputs):
            return {"status": "ERROR", "reason": "One or more input entities unresolved."}

        # 1. Test Single Multiplications / Divisions
        if len(resolved_inputs) == 2:
            a, b = resolved_inputs[0], resolved_inputs[1]
            
            # Try A * B
            prod = [x + y for x, y in zip(a.dimensions, b.dimensions)]
            if prod == target_entity.dimensions:
                return self._build_proof(f"{target_entity.name} = {a.name} · {b.name}", 
                                          [a, b], target_entity, "multiplication")

            # Try A / B
            div1 = [x - y for x, y in zip(a.dimensions, b.dimensions)]
            if div1 == target_entity.dimensions:
                return self._build_proof(f"{target_entity.name} = {a.name} / {b.name}", 
                                          [a, b], target_entity, "division")

            # Try B / A
            div2 = [y - x for x, y in zip(a.dimensions, b.dimensions)]
            if div2 == target_entity.dimensions:
                return self._build_proof(f"{target_entity.name} = {b.name} / {a.name}", 
                                          [b, a], target_entity, "division")

        # 2. Test Quadratic Forms (e.g. Mass * Speed^2)
        if len(resolved_inputs) == 2:
            a, b = resolved_inputs[0], resolved_inputs[1]
            # A * B^2
            quad1 = [x + 2*y for x, y in zip(a.dimensions, b.dimensions)]
            if quad1 == target_entity.dimensions:
                return self._build_proof(f"{target_entity.name} = {a.name} · {b.name}²", 
                                          [a, b], target_entity, "quadratic_composition")

        return {
            "status": "UNRESOLVED",
            "reason": f"No direct conservation pathway found connecting inputs to {target_entity.name}.",
            "target_dims": target_entity.dims_str()
        }

    def verify_equation(self, lhs_expr: str, rhs_expr: str) -> Dict[str, Any]:
        """
        Audits a proposed physical relation for dimensional balance,
        Leech symmetry consistency, and catches mod-2 false positive traps.
        """
        # Parse simple compositions like "mass * speed * speed" or "force * length"
        def parse_side(expr: str) -> Optional[List[int]]:
            tokens = [t.strip() for t in re.split(r'[*·/]', expr) if t.strip()]
            ops = re.findall(r'[*·/]', expr)
            if not tokens: return None
            
            ent = self.resolve_entity(tokens[0])
            if not ent: return None
            cur_dims = list(ent.dimensions)

            for op, tok in zip(ops, tokens[1:]):
                e_next = self.resolve_entity(tok)
                if not e_next: return None
                if op in ("*", "·"):
                    cur_dims = [x + y for x, y in zip(cur_dims, e_next.dimensions)]
                elif op == "/":
                    cur_dims = [x - y for x, y in zip(cur_dims, e_next.dimensions)]
            return cur_dims

        lhs_dims = parse_side(lhs_expr)
        rhs_dims = parse_side(rhs_expr)

        if not lhs_dims or not rhs_dims:
            return {"status": "ERROR", "reason": "Failed to parse one of the expressions."}

        exact_match = (lhs_dims == rhs_dims)
        mod2_match = ([x % 2 for x in lhs_dims] == [y % 2 for y in rhs_dims])
        mod2_trap = (mod2_match and not exact_match)

        lhs_ent = GeometricEntity("LHS", "LHS", lhs_dims)
        rhs_ent = GeometricEntity("RHS", "RHS", rhs_dims)

        return {
            "status": "VERIFIED_TRUE" if exact_match else "VERIFIED_FALSE",
            "equation": f"{lhs_expr} = {rhs_expr}",
            "lhs_dims": lhs_ent.dims_str(),
            "rhs_dims": rhs_ent.dims_str(),
            "mod2_trap_prevented": mod2_trap,
            "nrci_lhs": lhs_ent.nrci,
            "nrci_rhs": rhs_ent.nrci
        }

    def _build_proof(self, formula: str, inputs: List[GeometricEntity], target: GeometricEntity, mode: str) -> Dict[str, Any]:
        """Builds a structured deduction artifact."""
        return {
            "status": "DEDUCED_VALID",
            "formula": formula,
            "target": target.name,
            "target_dims": target.dims_str(),
            "input_entities": [i.name for i in inputs],
            "mode": mode,
            "proof_steps": [
                f"1. Target Quantity: {target.name} has 7D basis [{target.dims_str()}].",
                f"2. Input Quantities: {', '.join(f'{i.name} [{i.dims_str()}]' for i in inputs)}.",
                f"3. Group Action: Invariant balance established in (Z^7, +).",
                f"4. Topological Anchor: Target holds NRCI = {target.nrci:.4f} in Leech manifold."
            ]
        }

    def export_reasoning_scene(self, active_entities: List[GeometricEntity], filepath: str = "scene_3d.json"):
        """Exports the active reasoning graph directly to 3D Cartesian coordinates."""
        spheres, lines, pos_map = [], [], {}

        for e in active_entities:
            v = e.vector_24
            x = (sum(v[0:8]) - 4) * 2.0
            y = (sum(v[8:16]) - 4) * 2.0
            z = (sum(v[16:24]) - 4) * 2.0
            pos_map[e.name] = [x, y, z]

            # Color by Domain (Photometric = Gold, High-NRCI = Cyan)
            color = "#ffd700" if "J" in e.dims_str() else "#00ffff" if e.nrci >= 0.70 else "#ff00ff"
            spheres.append({
                "x": x, "y": y, "z": z,
                "r": 0.6 if e.name in ["energy", "force", "mass", "power"] else 0.4,
                "color": color,
                "label": f"{e.name} [{e.dims_str()}]"
            })

        # Connect pairs
        names = list(pos_map.keys())
        for i in range(len(names)):
            for j in range(i + 1, len(names)):
                lines.append({
                    "start": pos_map[names[i]],
                    "end": pos_map[names[j]],
                    "color": "#444444"
                })

        scene = {"spheres": spheres, "lines": lines}
        with open(filepath, "w", encoding="utf-8") as f:
            json.dump(scene, f, indent=2)
        print(f"  ✓ Exported 3D Geometric Reasoning Scene to '{filepath}'")
