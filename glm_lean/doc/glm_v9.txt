GF4_ADD = [[0,1,2,3],[1,0,3,2],[2,3,0,1],[3,2,1,0]]
GF4_SYM = {0:"0", 1:"1", 2:"ω", 3:"ω̄"}
ROW_W = [0, 1, 2, 3]

#: The complete 16-entry bijective table: F₂⁴ ↔ GF(4) × Z₄
COLUMN_TO_SHADOW: Dict[Tuple[int,int,int,int], Tuple[int,int]] = {}
SHADOW_TO_COLUMN: Dict[Tuple[int,int], Tuple[int,int,int,int]] = {}

for _val in range(16):
    _b = ((_val>>3)&1, (_val>>2)&1, (_val>>1)&1, _val&1)
    _s = 0
    for _r in range(4):
        if _b[_r]: _s = GF4_ADD[_s][ROW_W[_r]]
    _f = sum(1 for _c, (_sc, _) in COLUMN_TO_SHADOW.items() if _sc == _s)
    COLUMN_TO_SHADOW[_b] = (_s, _f)
    SHADOW_TO_COLUMN[(_s, _f)] = _b

#: The Z₄ fiber index maps to discrete S¹ versors (4th roots of unity).

#: Per Conway/Sloane: Co₀ = 2¹² : M₂₄
#:   2¹² = sign changes (the versors) → our Z₄ fibers

#: Per versors_and_fibers.txt:
#:   "Explicitly treat the Z₄ fiber index as the discrete Clifford/quaternionic
#:    phase group ⟨1, i, -1, -i⟩."

VERSOR_MAP = {0: 1+0j, 1: 0+1j, 2: -1+0j, 3: 0-1j}
VERSOR_NAMES = {0: "1", 1: "i", 2: "-1", 3: "-i"}


def fiber_versors(fiber_keys: List[int]) -> List[complex]:
    """Convert MOG fiber keys to quaternionic versor phases."""
    return [VERSOR_MAP[k] for k in fiber_keys]


def versor_phase(fiber_keys: List[int]) -> float:
    """Total phase (radians) of the versor product."""
    product = 1+0j
    for k in fiber_keys:
        product *= VERSOR_MAP[k]
    return cmath.phase(product) if product != 0 else 0.0


def versor_product(fiber_keys: List[int]) -> complex:
    """The net versor (rotation) of a concept."""
    product = 1+0j
    for k in fiber_keys:
        product *= VERSOR_MAP[k]
    return product

@dataclass
class MOGProjection:
    """The complete MOG projection of a 24-bit vector."""
    hex_scores: List[int]
    fiber_keys: List[int]
    syndrome_weight: int
    versors: List[complex] = field(init=False)
    phase: float = field(init=False)
    versor_prod: complex = field(init=False)
    tax: float = field(init=False)
    nrci: float = field(init=False)

    def __post_init__(self):
        self.versors = fiber_versors(self.fiber_keys)
        self.phase = versor_phase(self.fiber_keys)
        self.versor_prod = versor_product(self.fiber_keys)
        # TAX/NRCI computed on the snapped codeword (the "after" state)
        # For now, use the raw vector's tax
        hw = sum(1 for s in self.hex_scores if s != 0)  # approximate
        self.tax = hw * Y_FLOAT  # simplified
        self.nrci = B_FLOAT / (B_FLOAT + self.tax)


def project(v24: List[int]) -> MOGProjection:
    """Project 24-bit vector to MOG shadow with versor phases."""
    grid = [v24[i*6:(i+1)*6] for i in range(4)]
    scores, fibers = [], []
    for c in range(6):
        col = tuple(grid[r][c] for r in range(4))
        s, f = COLUMN_TO_SHADOW[col]
        scores.append(s)
        fibers.append(f)
    return MOGProjection(scores, fibers, GOLAY_ENGINE.syndrome_weight(v24))


def reconstruct(scores: List[int], fibers: List[int]) -> List[int]:
    """Reconstruct 24-bit from MOG shadow (lossless)."""
    grid = [[0]*6 for _ in range(4)]
    for c in range(6):
        col = SHADOW_TO_COLUMN[(scores[c], fibers[c])]
        for r in range(4):
            grid[r][c] = col[r]
    return grid[0] + grid[1] + grid[2] + grid[3]


@dataclass
class SnapResult:
    """The snap: (before, after, tax, phase_shift).

    THE DYNAMICS:
      σ(v) = H·v  is the syndrome (deviation from lawfulness)
      The snap corrects v to the nearest codeword.
      The phase shift = phase(before) - phase(after) is the geometric
      information of the correction — the "rotation" induced by resolving σ.

    Analogy with GA:
      GA: ∇F = J  (field residual)
      GLM: σ(v) = H·v  (syndrome residual)
      GA: solve ∇F = J for F
      GLM: snap(v) to resolve σ
      The phase shift is the geometric content of the resolution.
    """
    before: List[int]
    after: List[int]
    syndrome_weight: int
    correction_bits: List[int]
    phase_before: float
    phase_after: float
    phase_shift: float
    tax: float
    nrci: float

    @property
    def is_lawful(self) -> bool:
        return self.syndrome_weight == 0

    @property
    def status(self) -> str:
        if self.syndrome_weight == 0: return "lawful"
        if self.syndrome_weight <= 3: return "correctable"
        if self.syndrome_weight == 4: return "ambiguous (creative zone)"
        return "beyond covering radius"


def snap(v24: List[int]) -> SnapResult:
    """Snap to nearest codeword, computing the phase shift (dynamics)."""
    cw, meta = GOLAY_ENGINE.snap_to_codeword(v24)
    bits_changed = [i for i, (a, b) in enumerate(zip(v24, cw)) if a != b]

    # Versor phases before and after
    proj_before = project(v24)
    proj_after = project(cw)

    # TAX/NRCI on the snapped codeword (the lawful state)
    tax = float(LEECH_ENGINE.calculate_symmetry_tax(cw))
    nrci = float(LEECH_ENGINE.calculate_nrci(cw))

    return SnapResult(
        before=list(v24), after=list(cw),
        syndrome_weight=meta.get("syndrome_weight", 0),
        correction_bits=bits_changed,
        phase_before=proj_before.phase,
        phase_after=proj_after.phase,
        phase_shift=proj_before.phase - proj_after.phase,
        tax=tax, nrci=nrci,
    )

DIM_NAMES = ["L", "M", "T", "I", "Θ", "N", "J"]


@dataclass
class Concept:
    """A physical concept with dimensions, MOG encoding, versor phase, and snap.

    This combines:
    - Classical dimensional analysis (Z⁷, +) — the computation layer
    - Golay MOG encoding — the storage/transport layer
    - Versor fibers — the geometric orientation
    - Snap dynamics — the phase shift of correction
    """
    name: str
    dimensions: List[int]
    vector_24: List[int]
    mog: MOGProjection = field(init=False)
    snap_result: SnapResult = field(init=False)

    def __post_init__(self):
        self.mog = project(self.vector_24)
        self.snap_result = snap(self.vector_24)

    def dims_str(self) -> str:
        parts = []
        for n, e in zip(DIM_NAMES, self.dimensions):
            if e == 1: parts.append(n)
            elif e != 0: parts.append(f"{n}^{e}")
        return "·".join(parts) if parts else "dimensionless"


def compose(a: Concept, b: Concept, op: str = "multiply") -> Concept:
    """Compose via integer addition (deterministic, information-preserving)."""
    if op == "multiply":
        dims = [x + y for x, y in zip(a.dimensions, b.dimensions)]
        name = f"({a.name}·{b.name})"
    elif op == "divide":
        dims = [x - y for x, y in zip(a.dimensions, b.dimensions)]
        name = f"({a.name}/{b.name})"
    else:
        raise ValueError(f"Unknown operation: {op}")

    reality = [1 if dims[i] != 0 else 0 for i in range(6)]
    info = [dims[i] % 2 for i in range(5)] + [dims[6] % 2]
    activation = [1 if abs(dims[i]) > 1 else 0 for i in range(5)] + [1 if abs(dims[6]) > 1 else 0]
    potential = [1 if dims[i] < 0 else 0 for i in range(5)] + [1 if dims[6] < 0 else 0]
    vec = reality + info + activation + potential
    return Concept(name=name, dimensions=dims, vector_24=vec)


def check_equation(lhs: Concept, rhs: Concept) -> Dict[str, Any]:
    """Check dimensional homogeneity (exact integer, not mod-2)."""
    int_match = lhs.dimensions == rhs.dimensions
    mod2_match = [d % 2 for d in lhs.dimensions] == [d % 2 for d in rhs.dimensions]
    return {
        "accepted": int_match,
        "mod2_would_accept": mod2_match,
        "integer_fix_rejects": mod2_match and not int_match,
        "lhs_dims": lhs.dims_str(),
        "rhs_dims": rhs.dims_str(),
    }

# §5. LATTICE WALKS (winding invariants)
# ══════════════════════════════════════════════════════════════════════════════

@dataclass
class LatticeWalk:
    """A walk through concepts, accumulating versor phase."""
    steps: List[Dict[str, Any]] = field(default_factory=list)

    def add_concept(self, concept: Concept):
        """Add a concept to the walk."""
        self.steps.append({
            "name": concept.name,
            "dims": concept.dims_str(),
            "fibers": concept.mog.fiber_keys,
            "versors": [VERSOR_NAMES[f] for f in concept.mog.fiber_keys],
            "phase": concept.mog.phase,
            "phase_deg": math.degrees(concept.mog.phase),
            "syndrome": concept.mog.syndrome_weight,
            "tax": concept.snap_result.tax,
            "nrci": concept.snap_result.nrci,
            "phase_shift": concept.snap_result.phase_shift,
        })

    def total_phase(self) -> float:
        return sum(s["phase"] for s in self.steps)

    def winding_number(self) -> int:
        return round(self.total_phase() / (2 * math.pi))

    def describe(self) -> str:
        lines = [f"Lattice Walk ({len(self.steps)} steps):"]
        for i, s in enumerate(self.steps):
            lines.append(
                f"  {i}: {s['name']:<25} phase={s['phase_deg']:+7.1f}° "
                f"σ={s['syndrome']} TAX={s['tax']:.3f} NRCI={s['nrci']:.4f} "
                f"shift={math.degrees(s['phase_shift']):+7.1f}°"
            )
        lines.append(f"  Total phase: {math.degrees(self.total_phase()):+.1f}°")
        lines.append(f"  Winding: {self.winding_number()}")
        return "\n".join(lines)
