def string_to_24bit_hash(word: str) -> List[int]:
    """Creates a unique 24-bit seed from the word."""
    h = hashlib.sha256(word.lower().encode()).digest()
    bits = [(byte >> k) & 1 for byte in h for k in range(7, -1, -1)]
    return bits[:24]

def constrained_domain_snap(word: str, domain: str) -> dict:
    """
    Finds the nearest perfect Golay codeword that strictly obeys

    all_codewords = GOLAY_ENGINE.get_all_codewords()
    
    best_cw = None
