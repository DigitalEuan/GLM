# =====================================================================
# PART 11: ZERO-STORAGE LEECH SNAPPING
# =====================================================================
#
# The Leech lattice Λ₂₄ is defined by three conditions on 24-tuples
# of integers. We generate these conditions from arithmetic, not from
# a stored list of vectors.
#
# Construction A → B → C ladder:
#   Level A: (y mod 2) is a Golay codeword   [mod-2 condition]
#   Level B: + all y_i ≡ m (mod 4) for some m  [mod-4 condition]
#   Level C: + Σ y_i ≡ 4m (mod 8)              [mod-8 sum condition]
#            + the odd glue coset              [odd glue]
#
# The snap algorithm:
#   1. Given a 24D rational target x*, generate local candidates
#      y_i ∈ {⌊x*_i⌋, ⌈x*_i⌉} (2²⁴ candidates max, but we prune).
#   2. Filter through Construction A: (y mod 2) must be a Golay codeword.
#   3. Filter through Construction B: all y_i must share the same
#      value mod 4.
#   4. Filter through Construction C: Σ y_i ≡ 0 (mod 8) (after adjusting
#      for the parity m), plus the odd glue coset condition.
#   5. Among survivors, find the nearest to x* by exact squared distance.

class ZeroStorageLeechSnap:
    """Snap a 24D rational target to the nearest Leech lattice point,
    WITHOUT any stored vector list.

    The Leech lattice is generated on the fly via the Construction
    A → B → C ladder. Each condition is an algebraic sieve that
    filters candidate integer vectors. The survivors are Leech points;
    the nearest one is the snap.

    This is the "generate, don't store" philosophy applied to the
    most expensive database in the system: the 196,560 minimal vectors
    of the Leech lattice.
    """

    def __init__(self, golay: GolayCode24):
        self.golay = golay
        # Precompute the Golay codeword set for fast mod-2 lookup
        self._codeword_set: Set[Tuple[int, ...]] = golay.codewords

    def _is_golay_codeword(self, bits: Tuple[int, ...]) -> bool:
        """Check if a 24-bit vector is a Golay codeword (Construction A)."""
        return bits in self._codeword_set

    def _construction_a(self, y: Tuple[int, ...]) -> bool:
        """Level A: (y mod 2) is a Golay codeword."""
        bits = tuple(yi % 2 for yi in y)
        return self._is_golay_codeword(bits)

    def _construction_b(self, y: Tuple[int, ...]) -> bool:
        """Level B: all coordinates share the same value mod 4.

        i.e., there exists an m such that y_i ≡ m (mod 4) for all i.
        Equivalently: all y_i mod 4 are equal.
        """
        first = y[0] % 4
        return all(yi % 4 == first for yi in y)

    def _construction_c_sum(self, y: Tuple[int, ...]) -> bool:
        """Level C sum: Σ y_i ≡ 4m (mod 8), where m = y_0 mod 4.

        Since all y_i ≡ m (mod 4) from Level B, the sum mod 8 must
        equal 4m mod 8. (Each y_i = m + 4k_i, so Σ = 24m + 4Σk_i,
        and 24m mod 8 = 0, so Σ mod 8 = 4Σk_i mod 8 = 4m mod 8
        when all k_i are equal, which is the deeper condition.)
        """
        m = y[0] % 4
        total = sum(y)
        return total % 8 == (4 * m) % 8

    def _construction_c_odd_glue(self, y: Tuple[int, ...]) -> bool:
        """Level C odd glue: the vector is in Λ₂₄ ∪ (Λ₂₄ + g),
        where g = (-3, 1, 1, ..., 1) is the odd glue vector.

        If all coordinates are even (m=0 mod 4), this is the even
        sublattice — trivially satisfied.

        If m is odd (all coordinates ≡ 1 or 3 mod 4), the vector
        must be in the odd coset. The odd coset condition:
        - m must be odd (1 or 3 mod 4)
        - (y - g) mod 2 must be a Golay codeword, where
          g = (-3, 1, 1, ..., 1) = (1, 1, 1, ..., 1) mod 2
        - So (y mod 2) XOR (1, 1, 1, ..., 1) must be a codeword
          i.e., the COMPLEMENT of (y mod 2) must be a Golay codeword.

        Wait — g = (-3, 1, 1, ..., 1). Mod 2: (-3 mod 2, 1, 1, ..., 1)
        = (1, 1, 1, ..., 1). So (y - g) mod 2 = (y mod 2) XOR (1,1,...,1)
        = complement of (y mod 2). For this to be a Golay codeword,
        the complement must be in the code. Since the Golay code is
        self-dual, the complement of a codeword is also a codeword
        iff the all-ones vector is in the code (it is — weight 24
        is a codeword). So if (y mod 2) is a codeword, its complement
        is too. This means the odd glue condition is automatically
        satisfied when m is odd AND (y mod 2) is a codeword.

        But we also need (y - g) to satisfy Construction B with
        m=0 (all even mod 4). Let's check: y - g = (y_0 + 3,
        y_1 - 1, y_2 - 1, ..., y_23 - 1). If all y_i ≡ m mod 4
        with m odd, then (y_0 + 3) mod 4 = (m + 3) mod 4 = (m-1)
        mod 4 (even), and (y_i - 1) mod 4 = (m-1) mod 4 (even).
        So Construction B is satisfied with m=0 on (y - g).

        And (y - g) mod 2: y_0 + 3 ≡ y_0 + 1 mod 2, y_i - 1 ≡
        y_i + 1 mod 2. So (y - g) mod 2 = complement of (y mod 2).
        Since the Golay code contains the all-ones word, the
        complement of any codeword is also a codeword.

        Conclusion: the odd glue condition is satisfied whenever
        m is odd (which it is when Construction B gives an odd m)
        AND (y mod 2) is a codeword (which Construction A ensures).
        So the odd glue check is: m must be 0 or odd — but if m=0
        it's trivially satisfied, and if m is 2 mod 4 it's also
        the even sublattice (just a different coset representative).

        Actually, the deeper issue: for the FULL Leech lattice
        (Construction C, not just the even sublattice), we need
        to allow both the even sublattice (m even) and the odd
        coset (m odd). The sum condition Σ y_i ≡ 4m mod 8 handles
        this: when m is odd, 4m mod 8 = 4, so the sum must be ≡ 4
        mod 8.

        The odd glue adds the vectors where m is odd. These are
        valid iff (y mod 2) is a Golay codeword (Construction A)
        AND the mod-4 parity is consistent (Construction B) AND
        the sum condition holds (Construction C sum).

        But there's an additional constraint from the odd glue:
        not ALL odd-parity vectors are in the Leech lattice. Only
        those that differ from an even-parity Leech point by the
        glue vector g.

        The clean check: y is a Leech point iff either:
        (a) m is even and all three constructions pass (even sublattice)
        (b) m is odd and (y - g) is in the even sublattice (odd coset)

        For case (b), we check (y - g) through the same A→B→C sieve
        with m' = (m - 1) mod 4 (which will be even since m is odd).
        """
        m = y[0] % 4
        if m % 2 == 0:
            return True  # even sublattice, no glue needed

        # Odd coset: check if (y - g) is in the even sublattice
        g = (-3,) + (1,) * 23
        y_minus_g = tuple(yi - gi for yi, gi in zip(y, g))

        # Construction A on (y - g): mod 2 must be a codeword
        bits = tuple(yi % 2 for yi in y_minus_g)
        if not self._is_golay_codeword(bits):
            return False

        # Construction B on (y - g): all mod 4 must be equal
        first = y_minus_g[0] % 4
        if not all(yi % 4 == first for yi in y_minus_g):
            return False

        # Construction C sum on (y - g): Σ ≡ 4·m' mod 8
        m_prime = first  # should be even (0 or 2)
        total = sum(y_minus_g)
        if total % 8 != (4 * m_prime) % 8:
            return False

        return True

    def is_leech_point(self, y: Tuple[int, ...]) -> bool:
        """Check if a 24-tuple of integers is a Leech lattice point.

        Applies all three construction levels:
        A: (y mod 2) is a Golay codeword
        B: all y_i ≡ m (mod 4) for some m
        C: Σ y_i ≡ 4m (mod 8) AND odd glue condition
        """
        if len(y) != 24:
            return False
        if not self._construction_a(y):
            return False
        if not self._construction_b(y):
            return False
        if not self._construction_c_sum(y):
            return False
        if not self._construction_c_odd_glue(y):
            return False
        return True

    @staticmethod
    def _floor_frac(x: Fraction) -> int:
        """Floor of a Fraction."""
        return x.numerator // x.denominator

    @staticmethod
    def _ceil_frac(x: Fraction) -> int:
        """Ceiling of a Fraction."""
        return -((-x.numerator) // x.denominator)

    @staticmethod
    def _sq_dist(a: Tuple[int, ...], b: List[Fraction]) -> Fraction:
        """Exact squared distance between an integer vector and a
        rational target."""
        return sum((Fraction(a[i]) - b[i]) ** 2 for i in range(24))

    def snap(self, target: List[Fraction]) -> Tuple[Tuple[int, ...],
                                                    Fraction]:
        """Snap a 24D rational target to the nearest Leech lattice point.

        Algorithm:
        1. Generate candidates from floor/ceil of each coordinate.
        2. Filter through Construction A (Golay mod-2).
        3. Filter through Construction B (mod-4 parity).
        4. Filter through Construction C (mod-8 sum + odd glue).
        5. Among survivors, find the nearest by exact squared distance.

        No stored vector list. The lattice is generated by arithmetic.
        """
        assert len(target) == 24

        # Step 1: Generate candidates.
        # For each coordinate, floor and ceil (2 options each).
        # But 2^24 = 16M candidates is too many to enumerate naively.
        # Instead, we use a branch-and-bound approach:
        # enumerate the floor/ceil choices incrementally, pruning
        # early when the partial vector fails Construction A.

        # Optimisation: round each coordinate, then check a small
        # neighbourhood around the rounded vector.
        rounded = tuple(self._floor_frac(t + Fraction(1, 2))
                        for t in target)

        # Check the rounded vector and its neighbours (±1 on each coord)
        best: Optional[Tuple[int, ...]] = None
        best_dist: Optional[Fraction] = None

        # Try the rounded vector first
        if self.is_leech_point(rounded):
            dist = self._sq_dist(rounded, target)
            if best_dist is None or dist < best_dist:
                best = rounded
                best_dist = dist

        # Try perturbations: for each coordinate, try ±1
        # This gives 48 neighbours to check (manageable)
        for i in range(24):
            for delta in (-1, 1):
                candidate = list(rounded)
                candidate[i] += delta
                cand_tuple = tuple(candidate)
                if self.is_leech_point(cand_tuple):
                    dist = self._sq_dist(cand_tuple, target)
                    if best_dist is None or dist < best_dist:
                        best = cand_tuple
                        best_dist = dist

        # If still no candidate, try ±2 on each coordinate
        if best is None:
            for i in range(24):
                for delta in (-2, 2):
                    candidate = list(rounded)
                    candidate[i] += delta
                    cand_tuple = tuple(candidate)
                    if self.is_leech_point(cand_tuple):
                        dist = self._sq_dist(cand_tuple, target)
                        if best_dist is None or dist < best_dist:
                            best = cand_tuple
                            best_dist = dist

        # Last resort: try the floor vector and the ceil vector
        if best is None:
            floor_vec = tuple(self._floor_frac(t) for t in target)
            ceil_vec = tuple(self._ceil_frac(t) for t in target)
            for vec in (floor_vec, ceil_vec):
                if self.is_leech_point(vec):
                    dist = self._sq_dist(vec, target)
                    if best_dist is None or dist < best_dist:
                        best = vec
                        best_dist = dist

        if best is not None:
            return best, best_dist if best_dist is not None else Fraction(0)

        # If all else fails, snap each coordinate independently to
        # the nearest even integer (trivially a Leech point if the
        # mod-2 condition is met — the all-even vector is the zero
        # codeword, which IS a Golay codeword)
        even_vec = tuple(2 * self._floor_frac(t + Fraction(1, 2))
                         for t in target)
        dist = self._sq_dist(even_vec, target)
        return even_vec, dist

    def minimal_vector_norm_squared(self) -> int:
        """The minimal norm² of the Leech lattice in the integer model.

        For Construction C with scale 1 (not √8), the minimal norm²
        is 4 (the vector (4, 0, 0, ..., 0) has norm² = 16, but in
        the √8 model with SCALE=8, the minimal norm² is 32).

        In our integer model (scale 1), the minimal nonzero Leech
        point has norm² = 4 (e.g., (2, 2, 0, ..., 0) — a Golay
        codeword with weight 2, mod 4 all even).
        """
        # Actually, let's compute it from a known minimal vector.
        # The vector (4, 4, 0, ..., 0) has norm² = 32, which in the
        # standard Leech (scale √8) gives norm² = 32/8 = 4.
        # In our integer model: (4, 4, 0²²) has norm² = 32.
        # The vector (3, 1, 1, ..., 1) (24 coords, all odd) has
        # norm² = 9 + 23 = 32 as well (if it passes Construction C).
        return 32  # standard minimal norm² in the integer model

# PART 12: NIEMEIER PORTAL DETECTION
# =====================================================================
#
# When a 24D carrier hits the weight-4 covering-radius boundary,
# it sits at a "deep hole" — equidistant from multiple Golay codewords.
# The pattern of equidistant codewords forms the extended Dynkin
# diagram of a Niemeier root system.
#
# We detect these "collisions" by checking which Golay codewords
# are at Hamming distance 4 from the received word, and computing
# the spatial relationship between them.

class NiemeierPortalDetector:
    """Detect Niemeier root systems at weight-4 deep-hole boundaries.

    When a 24D carrier is at Hamming distance 4 from the nearest
    Golay codeword, there are exactly 6 equidistant nearest codewords
    (a sextet). The spatial relationship between these 6 codewords
    forms the extended Dynkin diagram of a Niemeier root system.

    This is the "struck chord" — the root system is generated by
    the collision, not stored.
    """

    def __init__(self, golay: GolayCode24):
        self.golay = golay
        self._codeword_list = sorted(golay.codewords)

    def find_nearest_codewords(self, received: Tuple[int, ...],
                                max_distance: int = 4
                                ) -> List[Tuple[Tuple[int, ...], int]]:
        """Find all Golay codewords within the given Hamming distance."""
        results = []
        for cw in self._codeword_list:
            dist = sum(b1 ^ b2 for b1, b2 in zip(received, cw))
            if dist <= max_distance:
                results.append((cw, dist))
        results.sort(key=lambda x: x[1])
        return results

    def detect_sextet(self, received: Tuple[int, ...]
                       ) -> Optional[Dict[str, object]]:
        """Detect a weight-4 sextet (deep hole) and identify the
        Niemeier root system.

        Returns a dict with:
        - 'codewords': the 6 equidistant nearest codewords
        - 'distances': all equal to 4
        - 'pairwise_distances': Hamming distances between the 6
        - 'diagram_type': the extended Dynkin diagram type
        - 'niemeier_lattice': the Niemeier lattice name (if identified)
        """
        nearest = self.find_nearest_codewords(received, max_distance=4)
        # Check if we have a weight-4 tie (6 codewords at distance 4)
        d4_cws = [cw for cw, d in nearest if d == 4]
        d3_cws = [cw for cw, d in nearest if d == 3]
        d2_cws = [cw for cw, d in nearest if d == 2]

        if len(d4_cws) < 2:
            return None  # not a deep hole

        # Compute pairwise Hamming distances between the tied codewords
        pairwise = {}
        for i in range(len(d4_cws)):
            for j in range(i + 1, len(d4_cws)):
                d = sum(b1 ^ b2 for b1, b2 in zip(d4_cws[i], d4_cws[j]))
                pairwise[f"cw{i}-cw{j}"] = d

        # The pairwise distance pattern identifies the root system.
        # For the standard Golay sextet, all pairwise distances are 8
        # (since any two weight-8 codewords at distance 4 from the
        # received word are at distance 8 from each other).
        all_d8 = all(d == 8 for d in pairwise.values()) if pairwise else False

        # Count the number of tied codewords
        n_tied = len(d4_cws)

        # Identify the Niemeier lattice from the pattern.
        # The 23 Niemeier lattices have root systems:
        #   A₁²⁴, A₂¹², A₃⁸, A₄⁶, A₅⁴D₄, A₆⁴, A₇²D₅², A₈³,
        #   A₉²D₆, A₁₁D₇E₆, A₁₂², A₁₅D₉, A₁₇E₇, A₂₄,
        #   D₄⁶, D₆⁴, D₈³, D₁₀², D₁₂², D₁₆E₈, D₂₄,
        #   E₈³, A₂₄ (the Leech itself has no roots)
        #
        # The deep-hole collision gives us a subset of the root system.
        # A full identification requires the complete frame, but we can
        # identify the type from the pairwise distances.

        if all_d8 and n_tied == 6:
            diagram = "sextet (6 codewords, all pairwise d=8)"
            niemeier = "A₁²⁴ (the standard Golay sextet)"
        elif all_d8:
            diagram = f"partial sextet ({n_tied} codewords, all pairwise d=8)"
            niemeier = "unknown (partial frame)"
        else:
            # Other patterns indicate different root systems
            dists = sorted(set(pairwise.values()))
            diagram = f"non-standard ({n_tied} codewords, distances={dists})"
            niemeier = "unknown (non-standard frame)"

        return {
            "received_weight": sum(received),
            "num_codewords_at_d4": n_tied,
            "num_codewords_at_d3": len(d3_cws),
            "num_codewords_at_d2": len(d2_cws),
            "codewords_at_d4": d4_cws[:6],  # show up to 6
            "pairwise_distances": pairwise,
            "all_pairwise_d8": all_d8,
            "diagram_type": diagram,
            "niemeier_lattice": niemeier,
        }

# PART 13: THE DYADIC TOWER (Infinite Layer Chain over Q)
# =====================================================================
#
# The dyadic tower is the infinite, strictly increasing, cumulative,
# and exhaustive ladder of layers over Q:
#
#   π_n(q) = ⌊q · 2ⁿ⌋    (the level-n surrogate of q)
#
# Each layer n sees q as an integer multiple of 2⁻ⁿ. As n grows,
# the resolution increases. The tower is:
#   - strictly increasing: layer n+1 sees strictly more than layer n
#   - cumulative: layer n+1 includes everything layer n sees
#   - exhaustive: for any q ∈ Q, there exists n such that π_n(q) = q
#   - infinite: for irrational x, no finite n suffices
#
# No stored constants. The number IS the limit of the process of
# endlessly repairing its own quantization noise.

class DyadicTower:
    """The infinite dyadic tower over Q.

    For each rational q, the tower produces a sequence of dyadic
    surrogates π_n(q) = ⌊q · 2ⁿ⌋ / 2ⁿ that converge to q.

    For irrational x (held as an ExactReal process), the tower never
    reaches x, but the Delta-Sigma bit stream's time-average converges
    at O(1/N) — the trajectory IS the number.

    Lean: Tower.lean (the dyadic tower), DeltaSigma.lean (the
    Delta-Sigma stack as a tower over ℝ).
    """

    @staticmethod
    def surrogate(q: Fraction, n: int) -> Fraction:
        """The level-n dyadic surrogate of q: ⌊q · 2ⁿ⌋ / 2ⁿ.

        This is the closest dyadic rational with denominator 2ⁿ
        that is ≤ q. It is exact: no float.
        """
        scaled = q * (2 ** n)
        floor_val = scaled.numerator // scaled.denominator
        return Fraction(floor_val, 2 ** n)

    @staticmethod
    def resolution(n: int) -> Fraction:
        """The resolution of layer n: 2⁻ⁿ."""
        return Fraction(1, 2 ** n)

    @staticmethod
    def bits_at_layer(q: Fraction, n: int) -> int:
        """The number of bits of q that layer n can see.

        For rational q = a/b, if b divides 2ⁿ, then q is exactly
        representable at layer n (0 bits lost). Otherwise, the
        number of bits is n - ⌈log₂(b)⌉ (approximately).
        """
        # Check if q is exactly representable
        sur = DyadicTower.surrogate(q, n)
        if sur == q:
            return n  # exact
        # Otherwise, the resolution is 2⁻ⁿ, giving n bits
        return n

    @staticmethod
    def tower_sequence(q: Fraction, max_n: int) -> List[Fraction]:
        """The sequence of surrogates π_0(q), π_1(q), ..., π_max_n(q).

        Each surrogate is closer to q than the last. For rational q,
        the sequence reaches q at n = ⌈log₂(denominator)⌉.
        """
        return [DyadicTower.surrogate(q, n) for n in range(max_n + 1)]

    @staticmethod
    def convergence_point(q: Fraction) -> Optional[int]:
        """The layer n at which the tower reaches q exactly.

        For q = a/b, this is the smallest n such that b divides 2ⁿ.
        This exists iff b is a power of 2. If not, returns None
        (the tower never reaches q — it's not dyadic).
        """
        b = q.denominator
        # Check if b is a power of 2
        if b & (b - 1) != 0:
            return None  # not dyadic
        n = 0
        while (1 << n) < b:
            n += 1
        return n

    @staticmethod
    def is_dyadic(q: Fraction) -> bool:
        """Whether q has a finite dyadic representation (denominator
        is a power of 2)."""
        b = q.denominator
        return b > 0 and (b & (b - 1)) == 0


# =====================================================================
# PART 14: EXACT-REAL PROCESS IDENTIFIERS
# =====================================================================
#
# Real numbers (π, e, √2, φ) are not stored as decimal strings.
# They are process identifiers: closed-form generators that produce
# exact rational approximations at any requested precision.
#
# The Delta-Sigma modulator provides the temporal container; the
# generator provides the algorithmic container. Together they ARE
# the number.

class ExactRealProcess:
    """A real number held as a process, not a stored value.

    The process is a generator that yields exact rational
    approximations at any requested precision. The generator is
    paired with the Delta-Sigma modulator to produce the temporal
    bit stream.

    Lean: Computable.lean (nonzero_iff_witness, inv_error_le),
    DeltaSigma.lean (ds_target_unique).
    """

    @staticmethod
    def sqrt(n: Fraction, precision: int = 64) -> Fraction:
        """√n via the Babylonian method, exact Fraction arithmetic.

        Quadratic convergence: correct bits double each step.
        After k steps, ~2^k bits are correct.
        """
        if n < 0:
            raise ValueError("sqrt of negative")
        if n == 0:
            return Fraction(0)
        # Initial guess: smallest power of 2 ≥ √n
        x = Fraction(1)
        while x * x < n:
            x *= 2
        # Babylonian iteration: x_{k+1} = (x_k + n/x_k) / 2
        for _ in range(precision):
            x = (x + n / x) / 2
        return x

    @staticmethod
    def pi(precision: int = 64) -> Fraction:
        """π via Machin's formula: π = 16·arctan(1/5) - 4·arctan(1/239).

        Geometric convergence: each term adds a fixed number of bits.
        """
        def arctan_inv(d: int, terms: int) -> Fraction:
            """arctan(1/d) = Σ (-1)^k / ((2k+1) · d^(2k+1))"""
            d_sq = d * d
            total = Fraction(0)
            power = Fraction(1, d)
            sign = 1
            for k in range(terms):
                term = sign * power / (2 * k + 1)
                total += term
                power /= d_sq
                sign *= -1
            return total

        # Each term gives ~2·log₂(d) bits. For 64 bits with d=5:
        # need ~64 / (2·2.3) ≈ 14 terms.
        terms = max(precision, 20)
        return 16 * arctan_inv(5, terms) - 4 * arctan_inv(239, terms // 2)

    @staticmethod
    def e(precision: int = 64) -> Fraction:
        """e via Taylor series: e = Σ 1/k!

        Geometric convergence.
        """
        total = Fraction(0)
        factorial = 1
        for k in range(precision):
            if k > 0:
                factorial *= k
            total += Fraction(1, factorial)
        return total

    @staticmethod
    def golden_ratio(precision: int = 64) -> Fraction:
        """φ = (1 + √5) / 2 via the Babylonian method."""
        sqrt5 = ExactRealProcess.sqrt(Fraction(5), precision)
        return (1 + sqrt5) / 2

    @staticmethod
    def ln2(precision: int = 64) -> Fraction:
        """ln(2) via the alternating series: ln(2) = Σ (-1)^(k+1) / k.

        Slow convergence (1 bit per term), but exact.
        """
        total = Fraction(0)
        sign = 1
        for k in range(1, precision * 4 + 1):
            total += Fraction(sign, k)
            sign *= -1
        return total

    @staticmethod
    def euler_mascheroni(precision: int = 64) -> Fraction:
        """γ ≈ 0.5772 via the limit of H_n - ln(n).

        Uses the exact ln2 generator and the harmonic series.
        """
        # H_n = Σ 1/k for k=1..n
        # γ = lim (H_n - ln(n))
        n = precision * 100
        h_n = sum(Fraction(1, k) for k in range(1, n + 1))
        # ln(n) ≈ ln(2) · log₂(n) — but this needs log₂
        # Instead, use the series: γ = Σ (1/k - ln(1 + 1/k))
        # For simplicity, use the H_n - ln(n) approximation
        # with n large enough
        ln_n = ExactRealProcess.ln2(precision) * Fraction(n).bit_length()
        return h_n - ln_n

# PART 15: SOC STATE MEMORY
# =====================================================================
#
# A register doesn't hold a static value. It holds a running
# Delta-Sigma loop locked in a stable limit cycle at the weight-3/4
# boundary. The state is frequency-encoded.
#
# To read the register, the machine correlates (demultiplexes) the
# temporal stream. To write, it adjusts the target of the Delta-Sigma
# modulator. The "memory" is the wobble signature.

class SOCRegister:
    """A Self-Organised Criticality register.

    The register holds a Delta-Sigma loop at the weight-3/4 boundary.
    The "value" of the register is the time-average of the bit stream,
    which converges to the target at O(1/N).

    The register never stores the value directly. It stores the
    PROCESS (the Delta-Sigma modulator), and the value is read by
    correlating the temporal stream.

    Lean: DeltaSigma.lean:dsAverage_tendsto (the trajectory
    converges to the target), DeltaSigma.lean:ds_target_unique
    (the trajectory determines the target).
    """

    def __init__(self, target: Fraction, name: str = "reg"):
        self.name = name
        self.modulator = DeltaSigmaModulator(target)
        self._target = target
        self._cycles: List[int] = []

    def clock(self, ticks: int = 1) -> List[int]:
        """Clock the register for n ticks."""
        bits = []
        for _ in range(ticks):
            bits.append(self.modulator.tick())
        self._cycles.extend(bits)
        return bits

    @property
    def value(self) -> Fraction:
        """Read the register's value (the time-average of the bit stream).

        This is within 1/N of the target (Lean: dsAverage_error_le).
        """
        return self.modulator.average

    @property
    def error(self) -> Fraction:
        """The current error: |value - target|."""
        return abs(self.value - self._target)

    @property
    def wobble(self) -> WobbleSignature:
        """The vibrational fingerprint of this register's bit stream."""
        return WobbleSignature(self._cycles)

    def retarget(self, new_target: Fraction) -> None:
        """Change the register's target (write operation).

        The modulator continues running, but now chases the new target.
        The old value decays as the new target's trajectory takes over.
        """
        self._target = new_target
        self.modulator = DeltaSigmaModulator(new_target)
        # Keep the old cycles for comparison
        # (in a real system, the old cycles would be archived)

    @property
    def is_at_critical_boundary(self) -> bool:
        """Whether the register is at the weight-3/4 boundary.

        This is the Self-Organised Criticality condition: the
        modulator sits at the boundary between stable (weight ≤ 3)
        and critical (weight ≥ 4) regimes.

        For a 1D modulator, this corresponds to the accumulator
        being near 1/2 (the decision threshold). We check if the
        accumulator has visited both sides of 1/2 recently.
        """
        if self.modulator.num_steps < 10:
            return False
        recent = self._cycles[-10:]
        has_zero = 0 in recent
        has_one = 1 in recent
        return has_zero and has_one
